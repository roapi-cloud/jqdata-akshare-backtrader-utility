# 机器学习策略通用框架设计

## 一、策略类型概述

### 核心逻辑
机器学习在量化交易中的核心逻辑是:通过历史数据训练模型,学习因子与未来收益之间的非线性关系,从而预测股票的未来表现并进行选股或择时。

### ML在量化中的应用场景
1. **多因子选股**: 使用基本面/技术面因子预测股票未来收益排名 (最常见)
2. **市场择时**: 预测大盘走势决定仓位 (如SVR+RSRS择时)
3. **ETF轮动**: 预测ETF未来表现进行轮动 (如BiLSTM for ETF)
4. **涨停/回撤预测**: 预测特定形态股票的未来走势 (如LightGBM回撤二波)
5. **交易信号生成**: 强化学习生成买卖持有信号 (如DQN交易智能体)

### 风险特征
| 风险类型 | 说明 | 应对策略 |
|---------|------|---------|
| **过拟合** | 模型在训练集表现好但样本外失效 | 交叉验证、简化模型、增加训练数据 |
| **数据泄漏** | 使用了未来数据导致回测虚高 | 严格使用`avoid_future_data`、时间点隔离 |
| **模型衰减** | 市场风格变化导致模型失效 | 滚动训练、定期重训、模型集成 |
| **幸存者偏差** | 只使用当前存活股票训练 | 使用历史截面全量股票 |
| **流动性风险** | 选出股票无法成交 | 过滤涨跌停、停牌、限制成交量 |

### 3.1 机器学习工作流
```
数据准备 → 特征工程 → 标签定义 → 模型训练 → 模型评估 → 预测 → 组合构建 → 回测
   ↓           ↓           ↓           ↓           ↓          ↓         ↓         ↓
股票池过滤  因子获取    未来收益    训练/加载   交叉验证   得分排序   等权/优化  实盘模拟
           去极值      分类标签    模型        IC/IR     TopN选股   调仓执行   绩效分析
           中性化      回归标签    滚动更新    回测验证            风控过滤
           标准化
```

```python
class FeatureEngineer:
    """特征工程器 - 从聚宽策略中提取的通用模式"""
    
    def get_fundamental_features(self, stock_list: List[str], date: str) -> pd.DataFrame:
        """
        基本面特征 - 来自 get_fundamentals
        常用因子:
        - 估值类: EP, BP, SP, NCFP, PE, PB, PS
        - 盈利类: ROE, ROA, gross_profit_margin, net_profit_ttm
        - 成长类: revenue_growth, profit_growth, sales_growth
        - 质量类: asset_turnover, cash_ratio, current_ratio
        - 杠杆类: financial_leverage, debt_to_assets
        """
        q = query(
            valuation.code, valuation.market_cap, valuation.pe_ratio,
            valuation.pb_ratio, valuation.ps_ratio,
            indicator.roe, indicator.roa, indicator.gross_profit_margin,
            indicator.inc_revenue_year_on_year, indicator.inc_net_profit_year_on_year
        ).filter(valuation.code.in_(stock_list))
        df = get_fundamentals(q, date)
        return df
    
    def get_jqfactor_features(self, stock_list: List[str], date: str, 
                               factor_names: List[str]) -> pd.DataFrame:
        """
        聚宽因子库特征 - 来自 get_factor_values
        常用因子类别:
        - 情绪类: ARBR, VOL120, TVSTD6, DAVOL20
        - 质量类: SGAI, total_profit_to_cost_ratio, inventory_turnover_rate
        - 动量类: Price1Y, sharpe_ratio_120
        - 技术类: price_no_fq, MACD, RSI, BIAS
        - 风险类: std_1m, std_3m, std_FF3factor_12m
        """
        factor_data = get_factor_values(stock_list, factor_names, 
                                         end_date=date, count=1)
        df = pd.DataFrame(index=stock_list)
        for factor in factor_names:
            df[factor] = list(factor_data[factor].T.iloc[:, 0])
        return df
    
    def get_technical_features(self, stock_list: List[str], date: str) -> pd.DataFrame:
        """
        技术指标特征 - 来自 jqlib.technical_analysis
        常用指标: MACD, KDJ, RSI, BIAS, DMA, EMA, MA, MTM, OBV, BRAR
        """
        df = pd.DataFrame(index=stock_list)
        df['MACD'] = pd.Series(MACD(stock_list, date, SHORT=12, LONG=26, MID=9)[2])
        df['RSI'] = pd.Series(RSI(stock_list, date, N1=20))
        df['BIAS'] = pd.Series(BIAS(stock_list, date, N1=20)[0])
        return df
    
    def get_industry_features(self, stock_list: List[str], date: str) -> pd.DataFrame:
        """行业哑变量特征 - One-Hot编码"""
        df = pd.DataFrame(index=stock_list)
        industry_list = get_industries('sw_l1', date).index.tolist()
        for sector in industry_list:
            istocks = get_industry_stocks(sector, date)
            df[sector] = [1 if s in istocks else 0 for s in stock_list]
        return df
    
    def preprocess_features(self, df: pd.DataFrame, date: str) -> pd.DataFrame:
        """
        特征预处理 - 标准量化流程
        1. 去极值 (winsorize_med)
        2. 缺失值处理 (行业均值填充)
        3. 中性化 (neutralize - 行业+市值)
        4. 标准化 (standardlize)
        """
        # 去极值 - 中位数去极值法
        df = winsorize_med(df, scale=5, inf2nan=False, axis=0)
        
        # 缺失值处理 - 行业均值填充
        df = self._fill_nan_by_industry(df, date)
        
        # 中性化 - 行业和市值中性
        df = neutralize(df, how=['sw_l1', 'market_cap'], date=date, axis=0)
        
        # 标准化 - Z-Score标准化
        df = standardlize(df, axis=0)
        
        return df
    
    def _fill_nan_by_industry(self, df: pd.DataFrame, date: str) -> pd.DataFrame:
        """缺失值用行业均值填充"""
        # 简化实现: 用列均值填充
        df = df.fillna(df.median())
        return df
```

class LabelGenerator:
    """标签生成器 - 定义预测目标"""
    
    def generate_return_label(self, stock_list: List[str], start_date: str, 
                               end_date: str) -> pd.Series:
        """未来收益率标签 - 回归任务"""
        returns = {}
        for stock in stock_list:
            prices = get_price(stock, start_date=start_date, end_date=end_date,
                              frequency='daily', fields=['close'])
            if len(prices) > 0:
                returns[stock] = prices['close'].iloc[-1] / prices['close'].iloc[0] - 1
        return pd.Series(returns)
    
    def generate_classification_label(self, returns: pd.Series, 
                                        top_pct: float = 0.3,
                                        bottom_pct: float = 0.3) -> pd.Series:
        """
        分类标签 - 涨/跌/平
        常见做法: 取前后30%作为正负样本,中间40%作为噪声剔除
        """
        top_threshold = returns.quantile(1 - top_pct)
        bottom_threshold = returns.quantile(bottom_pct)
        
        labels = pd.Series(index=returns.index)
        labels[returns >= top_threshold] = 1  # 上涨
        labels[returns <= bottom_threshold] = 0  # 下跌
        # 中间部分剔除(噪声)
        labels = labels.dropna()
        return labels
    
    def generate_rank_label(self, stock_list: List[str], date: str, 
                             forward_days: int = 20) -> pd.Series:
        """排名标签 - 用于排序学习(Learning to Rank)"""
        # 计算未来N日收益并排名

        return returns.rank(ascending=False)

class ModelManager:
    """模型管理器 - 训练/加载/预测"""
    
    def train_classification(self, X_train, y_train, model_type: str = "xgboost",
                              params: Dict = None) -> Any:
        """训练分类模型"""
        if model_type == "xgboost":
            from xgboost import XGBClassifier
            default_params = {
                'max_depth': 3, 'learning_rate': 0.05, 'subsample': 0.8,
                'n_estimators': 100
            }
            params = params or default_params
            model = XGBClassifier(**params)
        elif model_type == "random_forest":
            from sklearn.ensemble import RandomForestClassifier
            default_params = {
                'n_estimators': 50, 'max_depth': None, 'min_samples_split': 2
            }
            params = params or default_params
            model = RandomForestClassifier(**params)
        elif model_type == "svm":
            from sklearn.svm import SVC
            model = SVC(C=0.1, probability=True)
        elif model_type == "lightgbm":
            import lightgbm as lgb
            model = lgb.LGBMClassifier(**(params or {}))
        
        model.fit(X_train, y_train)
        return model
    
    def train_regression(self, X_train, y_train, model_type: str = "svr",
                          params: Dict = None) -> Any:
        """训练回归模型"""
        if model_type == "svr":
            from sklearn.svm import SVR
            model = SVR(kernel='rbf', **(params or {}))
        elif model_type == "xgboost":
            from xgboost import XGBRegressor
            model = XGBRegressor(**(params or {}))
        
        model.fit(X_train, y_train)
        return model
    
    def predict_proba(self, model, X_test) -> np.ndarray:
        """预测概率 - 用于分类模型"""
        return model.predict_proba(X_test)[:, 1]  # 取正类概率
    
    def predict(self, model, X_test) -> np.ndarray:
        """预测值 - 用于回归模型"""
        return model.predict(X_test)
    
    def save_model(self, model, path: str):
        """保存模型"""
        import pickle
        with open(path, 'wb') as f:
            pickle.dump(model, f)
    
    def load_model(self, path: str) -> Any:
        """加载模型 - 聚宽平台使用 read_file"""
        import pickle
        model = pickle.loads(read_file(path))
        return model

class RollingTrainer:
    """
    滚动训练器 - 时序交叉验证避免数据泄漏
    来自策略: 64 基于XGBoost_6m滚动选股
    """
    
    def __init__(self, train_window: int = 6, test_window: int = 1):
        """
        Args:
            train_window: 训练窗口(月数)
            test_window: 测试窗口(月数)
        """
        self.train_window = train_window
        self.test_window = test_window
    
    def get_rolling_dates(self, end_date: str, count: int = 12) -> List[str]:
        """获取滚动日期序列"""
        trade_days = get_trade_days(end_date=end_date, count=count * 20)
        # 按月取样
        monthly_dates = trade_days[::20].tolist()
        return [str(d) for d in monthly_dates[-self.train_window-1:]]
    
    def rolling_train(self, dates: List[str], feature_func, label_func,
                       model_type: str = "xgboost") -> Any:
        """
        滚动训练流程:
        1. 对每个历史日期,获取特征和标签
        2. 拼接所有历史数据作为训练集
        3. 训练模型
        4. 用最新日期数据预测
        """
        train_data = pd.DataFrame()
        
        for date in dates[:-1]:  # 最后一个日期是测试集
            # 获取特征
            X = feature_func(date)
            # 获取标签(未来收益)
            y = label_func(date, dates[dates.index(date) + 1])
            
            # 拼接
            X['label'] = y

class PredictionSelector:
    """预测选股器 - 从预测到实际持仓"""
    
    def select_by_score(self, predictions: np.ndarray, stock_list: List[str],
                         top_n: int = 10) -> List[str]:
        """按预测得分选股 - TopN"""
        scores = pd.Series(predictions, index=stock_list)
        scores = scores.sort_values(ascending=False)
        return scores.head(top_n).index.tolist()
    
    def select_by_threshold(self, predictions: np.ndarray, stock_list: List[str],
                              threshold: float = 0.6) -> List[str]:
        """按概率阈值选股"""
        scores = pd.Series(predictions, index=stock_list)
        return scores[scores > threshold].index.tolist()
    
    def secondary_filter(self, stock_list: List[str], date: str,
                           filter_type: str = "market_cap",
                           top_n: int = 10) -> List[str]:
        """
        二次过滤 - 常见做法:
        1. 按市值排序选小市值 (小市值策略)
        2. 按EPS过滤 (盈利股票)
        3. 按ROA/ROE排序 (质量因子)
        """
        if filter_type == "market_cap":
            q = query(valuation.code, valuation.circulating_market_cap)\
                .filter(valuation.code.in_(stock_list))\
                .order_by(valuation.circulating_market_cap.asc())
            df = get_fundamentals(q, date)
            df = df[df['circulating_market_cap'] > 0]
            return list(df.code)[:top_n]
        elif filter_type == "eps":
            q = query(valuation.code, indicator.eps)\
                .filter(valuation.code.in_(stock_list), indicator.eps > 0)\
                .order_by(valuation.circulating_market_cap.asc())
            df = get_fundamentals(q, date)
            return list(df.code)[:top_n]
        elif filter_type == "roa":
            q = query(valuation.code, indicator.roa)\
                .filter(valuation.code.in_(stock_list))\
                .order_by(indicator.roa.desc())
            df = get_fundamentals(q, date)
            return list(df.code)[:top_n]
        
        return stock_list[:top_n]

class StockPoolFilter:
    """股票池过滤器 - 所有ML策略通用的过滤逻辑"""
    
    def filter_all(self, stock_list: List[str], context, 
                    exclude_new_days: int = 375) -> List[str]:
        """综合过滤"""
        stock_list = self.filter_st(stock_list)
        stock_list = self.filter_paused(stock_list)
        stock_list = self.filter_kcbj(stock_list)
        stock_list = self.filter_new_stock(stock_list, context, exclude_new_days)
        stock_list = self.filter_limit_up(stock_list, context)
        stock_list = self.filter_limit_down(stock_list, context)
        return stock_list
    
    def filter_st(self, stock_list: List[str]) -> List[str]:
        current_data = get_current_data()
        return [s for s in stock_list 
                if not current_data[s].is_st 
                and 'ST' not in current_data[s].name
                and '*' not in current_data[s].name
                and '退' not in current_data[s].name]
    
    def filter_paused(self, stock_list: List[str]) -> List[str]:
        current_data = get_current_data()
        return [s for s in stock_list if not current_data[s].paused]
    
    def filter_kcbj(self, stock_list: List[str]) -> List[str]:
        """过滤科创板(688)、北交所(4/8开头)、创业板(300)"""
        return [s for s in stock_list 
                if s[0] not in ('4', '8') 
                and s[:3] != '688' 
                and s[:3] != '300']
    
    def filter_new_stock(self, stock_list: List[str], context, 
                           days: int = 375) -> List[str]:
        yesterday = context.previous_date
        return [s for s in stock_list 
                if (yesterday - get_security_info(s).start_date).days >= days]
    
    def filter_limit_up(self, stock_list: List[str], context) -> List[str]:
        last_prices = history(1, unit='1m', field='close', security_list=stock_list)
        current_data = get_current_data()
        return [s for s in stock_list 
                if s in context.portfolio.positions.keys()
                or last_prices[s][-1] < current_data[s].high_limit]
    
    def filter_limit_down(self, stock_list: List[str], context) -> List[str]:
        last_prices = history(1, unit='1m', field='close', security_list=stock_list)
        current_data = get_current_data()
        return [s for s in stock_list 
                if s in context.portfolio.positions.keys()
                or last_prices[s][-1] > current_data[s].low_limit]

### 3.4 框架主类

```python
class MLTradingStrategy:
    """
    机器学习交易策略主类
    整合所有模块,提供完整的ML策略生命周期
    """
    
    def __init__(self, config: MLStrategyConfig):
        self.config = config
        self.feature_engineer = FeatureEngineer()
        self.label_generator = LabelGenerator()
        self.model_manager = ModelManager()
        self.rolling_trainer = RollingTrainer(config.train_window_months)
        self.prediction_selector = PredictionSelector()
        self.stock_pool_filter = StockPoolFilter()
        self.model = None
        
    def initialize(self, context):
        """策略初始化"""
        # 基础设置
        set_benchmark('000905.XSHG')
        set_option('use_real_price', True)
        set_option("avoid_future_data", True)
        set_slippage(FixedSlippage(0))
        set_order_cost(OrderCost(
            open_tax=0, close_tax=0.001, 
            open_commission=0.0003, close_commission=0.0003, 
            min_commission=5
        ), type='stock')
        log.set_level('order', 'error')
        
        # 全局变量
        g.stock_num = self.config.stock_num
        g.hold_list = []
        
        # 加载预训练模型 (如果有)
        if self.config.model_file_path:
            self.model = self.model_manager.load_model(self.config.model_file_path)
        
        # 设置定时任务
        run_daily(self._prepare_stock_list, '9:05')
        
        if self.config.rebalance_frequency == 'weekly':
            run_weekly(self._rebalance, self.config.rebalance_day, '9:30')
        elif self.config.rebalance_frequency == 'monthly':
            run_monthly(self._rebalance, self.config.rebalance_day, '9:30')
        
        run_daily(self._check_limit_up, '14:00')
    
    def _prepare_stock_list(self, context):
        """盘前准备"""
        g.hold_list = list(context.portfolio.positions.keys())
    
    def _rebalance(self, context):
        """
        核心调仓逻辑:
        1. 获取股票池
        2. 构建特征矩阵
        3. 训练/更新模型
        4. 预测得分
        5. 选股
        6. 执行调仓
        """
        yesterday = context.previous_date
        
        # Step 1: 获取初始股票池
        stock_list = self._get_initial_stock_pool(context)
        
        # Step 2: 构建特征矩阵
        X = self._build_feature_matrix(stock_list, yesterday)
        
        # Step 3: 训练或加载模型
        if self.model is None:
            self.model = self._train_model(context, X, stock_list, yesterday)
        elif self.config.training_mode == "rolling":
            self.model = self._train_model(context, X, stock_list, yesterday)
        
        # Step 4: 预测
        if hasattr(self.model, 'predict_proba'):
            scores = self.model.predict_proba(X)[:, 1]
        else:
            scores = self.model.predict(X)
        
        # Step 5: 选股
        selected = self.prediction_selector.select_by_score(
            scores, list(X.index), 
            top_n=int(len(X) * self.config.prediction_top_pct)
        )
        
        # 二次过滤
        if self.config.secondary_filter != "none":
            selected = self.prediction_selector.secondary_filter(
                selected, yesterday, self.config.secondary_filter, self.config.stock_num
            )
        
        # Step 6: 执行调仓
        self._execute_rebalance(context, selected)
    
    def _get_initial_stock_pool(self, context) -> List[str]:
        """获取初始股票池"""
        if self.config.stock_pool == "all_a":
            stocks = get_all_securities('stock', context.previous_date).index.tolist()
        elif self.config.stock_pool == "csi500":
            stocks = get_index_stocks('399905.XSHE', context.previous_date)
        elif self.config.stock_pool == "csi300":
            stocks = get_index_stocks('000300.XSHG', context.previous_date)
        else:
            stocks = get_all_securities('stock', context.previous_date).index.tolist()
        
        # 过滤
        stocks = self.stock_pool_filter.filter_all(stocks, context, 
                                                     self.config.exclude_new_days)
        return stocks
    
    def _build_feature_matrix(self, stock_list: List[str], date: str) -> pd.DataFrame:
        """构建特征矩阵"""
        # 获取因子值
        if self.config.factor_source in ["jqfactor", "mixed"]:
            X = self.feature_engineer.get_jqfactor_features(
                stock_list, date, self.config.factor_names
            )
        
        # 行业哑变量
        if self.config.use_industry_dummy:
            industry_df = self.feature_engineer.get_industry_features(stock_list, date)
            X = pd.concat([X, industry_df], axis=1)
        
        # 预处理
        X = self.feature_engineer.preprocess_features(X, date)
        
        return X.dropna()
    
    def _train_model(self, context, X, stock_list, date):
        """训练模型"""
        # 获取训练日期序列
        dates = self.rolling_trainer.get_rolling_dates(
            date, self.config.train_window_months + 1
        )
        
        # 构建训练数据
        train_data = pd.DataFrame()
        for d in dates[:-1]:
            X_hist = self._build_feature_matrix(stock_list, d)
            # 计算标签(未来收益)
            next_date = dates[dates.index(d) + 1]
            y = self.label_generator.generate_return_label(stock_list, d, next_date)
            
            X_hist['label'] = y
            X_hist = X_hist.dropna()
            
            if train_data.empty:
                train_data = X_hist
            else:
                train_data = pd.concat([train_data, X_hist])
        
        # 生成分类标签
        y_train = self.label_generator.generate_classification_label(
            train_data['label'],
            self.config.label_threshold_top,
            self.config.label_threshold_bottom
        )
        X_train = train_data.loc[y_train.index].drop('label', axis=1)
        
        # 训练
        model = self.model_manager.train_classification(
            X_train, y_train, self.config.model_type, self.config.model_params
        )
        
        return model
    
    def _execute_rebalance(self, context, target_list: List[str]):
        """执行调仓"""
        # 卖出不在目标列表的股票
        for stock in g.hold_list:
            if stock not in target_list:
                order_target_value(stock, 0)
        
        # 买入目标股票
        position_count = len(context.portfolio.positions)
        target_num = len(target_list)
        
        if target_num > position_count:
            value = context.portfolio.cash / (target_num - position_count)
            for stock in target_list:
                if context.portfolio.positions[stock].total_amount == 0:
                    order_target_value(stock, value)
```

class DQN(nn.Module):
