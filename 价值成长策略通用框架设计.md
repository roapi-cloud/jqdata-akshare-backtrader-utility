# 价值/成长策略通用框架设计

## 一、策略类型概述

### 核心逻辑
价值/成长策略的核心逻辑是**基于基本面分析，寻找被市场低估或具备持续成长能力的优质公司**。主要分为三大流派：

1. **价值投资(Value)**: 寻找低估值股票(低PE/PB/高股息率)，相信市场最终会纠正定价错误
2. **成长投资(Growth)**: 寻找高成长股票(高营收增长/利润增长/ROE改善)，相信成长性能持续转化为股价上涨
3. **GARP(Growth at Reasonable Price)**: 结合价值与成长，寻找成长性与估值匹配的股票(PEG指标为核心)

### 适用场景
- **价值策略**: 熊市末期/震荡市/牛市初期效果最佳，牛市末期容易跑输
- **成长策略**: 牛市/经济复苏期效果最佳，熊市容易大幅回撤
- **GARP策略**: 全市场适用，但需要更精细的因子筛选

### 风险特征
- **优势**: 长期收益稳定，回撤可控，逻辑清晰可解释
- **劣势**: 短期可能跑输市场，需要耐心持有；财报数据有滞后性
- **典型回撤**: 10%-30%（取决于是否加入择时/风控模块）

```python
        ms_list = list(ms_df.code[:int(len(ms_df)*percentile)])
        
        # 按市值降序
        q = query(valuation.code).filter(valuation.code.in_(ms_list)).order_by(

        2. 去极值(winsorize)
        3. 行业中性化(neutralize)
        4. 标准化(standardlize)
        5. 等权/加权求和
        6. 排序取前N
        """

            valuation.code,
            indicator.roe,
            indicator.roa,
            valuation.pe_ratio,
            valuation.pb_ratio
        ).filter(valuation.code.in_(stock_list)), date=date)
        
        df = df.set_index('code')
        
        # 去极值、中性化、标准化(39大资金优质股模式)
        for factor in criteria.get('factors', ['roe', 'roa']):
            df[factor] = winsorize(df[factor], qrange=[0.05, 0.95])
            df[factor] = neutralize(df[factor], how=['jq_l1', 'market_cap'], date=date)
            df[factor] = standardlize(df[factor])
        
        # 计算综合得分
        weights = criteria.get('weights', {factor: 1.0 for factor in criteria.get('factors', [])})
        df['score'] = sum(df[factor] * weights[factor] for factor in weights)
        df = df.sort_values('score', ascending=False)
        
        return list(df.index[:criteria.get('top_n', 10)])
```

```python
        for q_date in quarters_dates:

            fields=[indicator.code, indicator.roe], 
            watch_date=date, 
            count=quarters, 
            interval='1q'
        )
        
        # 计算加权平均
        df_grouped = df.groupby('code').apply(lambda x: x.reset_index()).roe.unstack()
        df_grouped['past_average'] = (0.1*df_grouped.iloc[:,0] + 0.2*df_grouped.iloc[:,1] + 
                                       0.3*df_grouped.iloc[:,2] + 0.4*df_grouped.iloc[:,3])
        df_grouped['now_average'] = (0.1*df_grouped.iloc[:,1] + 0.2*df_grouped.iloc[:,2] + 
                                      0.3*df_grouped.iloc[:,3] + 0.4*df_grouped.iloc[:,4])
        df_grouped['delta_average'] = df_grouped['now_average'] - df_grouped['past_average']
        df_grouped = df_grouped.dropna().sort_values('delta_average', ascending=False)
        
        return list(df_grouped.index[:int(len(df_grouped)*0.1)])
```

```python
class QualityScreener:
    """质量筛选器 - 财务健康度筛选"""
    
    def screen_by_roe(self, stock_list: List[str], date: str, 
                      min_roe: float = 0.10, quarters: int = 4) -> List[str]:
        """ROE筛选 - 连续多季度ROE达标
        
        典型参数: min_roe=0.10 (45价值投资+期货对冲, 60高股息V1.0用roic_ttm>0.08)
                 连续4季度ROE>市场均值 (35精选价值, 55价值投资改进版, 87穿越牛熊)
        """
        quarters_dates = self._get_stat_dates(date, quarters)
        
        result_set = None
        for q_date in quarters_dates:
            q = query(indicator.code).filter(
                indicator.code.in_(stock_list),
                indicator.roe > min_roe * 100
            )
            df = get_fundamentals(q, statDate=q_date)
            if result_set is None:
                result_set = set(df.code)
            else:
                result_set = result_set & set(df.code)
        
        return list(result_set) if result_set else []
    
    def screen_by_roe_above_market(self, stock_list: List[str], date: str,
                                    quarters: int = 4) -> List[str]:
        """ROE高于市场均值筛选 - 经典价值策略模式
        
        典型应用: 35精选价值策略, 55价值投资改进版, 87穿越牛熊
            近4个季度ROE都高于各自季度的市场均值
        """
        quarters_dates = self._get_stat_dates(date, quarters)
        
        result_set = None
        for q_date in quarters_dates:
            df = get_fundamentals(query(
                indicator.code, indicator.roe
            ).filter(indicator.code.in_(stock_list)), statDate=q_date)
            
            if len(df) > 0:
                roe_mean = df['roe'].mean()
                df = df[df['roe'] > roe_mean]
                if result_set is None:
                    result_set = set(df.code)
                else:
                    result_set = result_set & set(df.code)
        
        return list(result_set) if result_set else []
    
    def screen_by_roa(self, stock_list: List[str], date: str, 
                      min_roa: float = 0.05) -> List[str]:
        """ROA筛选 - 大市值价值策略核心条件
        
        典型参数: min_roa=0.15 (16大市值价值, 75大市值优化, 55大市值+邮箱)
                 min_roa=0.05 (45价值投资+期货对冲: roa>0.05*roe)
        """
        q = query(valuation.code).filter(
            valuation.code.in_(stock_list),
            indicator.roa > min_roa
        )
        return list(get_fundamentals(q).code)
    
    def screen_by_fcf(self, stock_list: List[str], date: str, 
                      years: int = 3, min_fcf: float = 0) -> List[str]:
        """连续正自由现金流筛选 - 经典价值策略核心条件
        
        典型参数: years=3 (35精选价值, 55价值投资改进版, 87穿越牛熊, 90大师策略)
        
        计算方式: FCF = 经营活动现金流净额 - 投资活动现金流出
                 或: FCF = net_operate_cash_flow + net_invest_cash_flow (注意符号)
        """
        y = date.year
        if date.month >= 5:
            years_list = [y-1, y-2, y-3][:years]
        else:
            years_list = [y-2, y-3, y-4][:years]
        
        result_set = None
        for year in years_list:
            q = query(
                cash_flow.code,
                cash_flow.net_operate_cash_flow,
                cash_flow.net_invest_cash_flow
            ).filter(cash_flow.code.in_(stock_list))
            df = get_fundamentals(q, statDate=str(year))
            
            if len(df) > 0:
                df['FCF'] = df['net_operate_cash_flow'] + df['net_invest_cash_flow']
                df = df[df['FCF'] > min_fcf]
                if result_set is None:
                    result_set = set(df.code)
                else:
                    result_set = result_set & set(df.code)
        
        return list(result_set) if result_set else []
    
    def screen_by_current_ratio(self, stock_list: List[str], date: str) -> List[str]:
        """流动比率筛选 - 财务结构健康度
        
        典型应用: 35精选价值, 55价值投资改进版, 87穿越牛熊
            最近一季流动比率 >= 市场平均值
        
        计算: 流动比率 = 流动资产合计 / 流动负债合计
        """
        df = get_fundamentals(query(
            balance.code,
            balance.total_current_assets,
            balance.total_current_liability
        ).filter(balance.code.in_(stock_list)), date=date)
        
        df = df.dropna()
        df = df[df['total_current_liability'] != 0]
        df['current_ratio'] = df['total_current_assets'] / df['total_current_liability']
        
        mean_ratio = df['current_ratio'].mean()
        df = df[df['current_ratio'] > mean_ratio]
        
        return list(df.code)
    
    def screen_by_debt_ratio(self, stock_list: List[str], date: str,
                              max_debt_ratio: float = 0.90) -> List[str]:
        """资产负债率筛选 - 财务杠杆控制
        
        典型参数: max_debt_ratio=0.90 (85价值策略mode1)
                 max_debt_ratio=0.75 (25 ROE-PB模型: 资产负债率<75分位)
        """
        df = get_fundamentals(query(
            balance.code,
            balance.total_liability,
            balance.total_assets
        ).filter(balance.code.in_(stock_list)), date=date)
        
        df = df.dropna()
        df['debt_ratio'] = df['total_liability'] / df['total_assets']
        df = df[df['debt_ratio'] < max_debt_ratio]
        
        return list(df.code)
```

```python
class VolatilityCalculator:
    """波动率计算器 - 用于低波策略和风险加权"""
    
    def get_historical_volatility(self, stock: str, date: str, period: int = 241) -> float:
        """历史波动率 - 年化收益率标准差
        
        典型应用: 05价值低波(period=241, 约1年交易日)
        """
        h = history(period, '1d', 'close', [stock], df=True)
        r = h.pct_change()[1:]
        return r.std().iloc[0]
    
    def get_volatility_weight(self, stock_list: List[str], date: str,
                               period: int = 241) -> pd.Series:
        """波动率倒数加权 - 低波策略核心
        
        典型应用: 05价值低波
        
        实现逻辑:
        1. 获取近period天收盘价
        2. 计算日收益率
        3. 计算收益率标准差(波动率)
        4. 取波动率倒数作为权重
        5. 归一化使权重和为1
        """
        h = history(period, '1d', 'close', stock_list, df=True)
        r = h.pct_change()[1:]
        
        # 波动率倒数
        w = 1.0 / r.std()
        weight = w / w.sum()
        
        return weight
```

```python
class ValueGrowthStrategy:
    """价值/成长策略框架主类"""

    def prepare_stock_list(self, context):
        """盘前准备 - 更新持仓列表和涨停列表"""
        self.hold_list = list(context.portfolio.positions.keys())
        
        # 获取昨日涨停列表(所有策略通用模式)
        if self.hold_list:
            df = get_price(self.hold_list, end_date=context.previous_date, 
                          frequency='daily', fields=['close', 'high_limit'], count=1)
            df = df[df['close'] == df['high_limit']]
            self.high_limit_list = list(df.code)
        else:
            self.high_limit_list = []
    
    def before_market_open(self, context):
        """开盘前运行 - 选股"""
        # 1. 获取股票池
        stock_list = self._get_stock_pool(context)
        
        # 2. 基础过滤(ST/停牌/次新/科创板)
        stock_list = self._filter_basic(stock_list, context)
        
        # 3. 价值筛选
        if self.config.use_pe_filter:
            stock_list = self.value_screener.screen_by_pe(
                stock_list, self.config.pe_min, self.config.pe_max)
        
        if self.config.use_pb_filter:
            stock_list = self.value_screener.screen_by_pb(
                stock_list, self.config.pb_min, self.config.pb_max)
        
        if self.config.use_dividend_filter:
            stock_list = self.value_screener.screen_by_dividend_yield(
                stock_list, context.previous_date, 
                self.config.dividend_years, self.config.dividend_top_pct)
        
        if self.config.use_peg_filter:
            stock_list = self.value_screener.screen_by_peg(
                stock_list, self.config.peg_min, self.config.peg_max)
        
        # 4. 成长筛选(可选)
        if self.config.use_revenue_growth:
            stock_list = self.growth_screener.screen_by_revenue_growth(
                stock_list, context.previous_date,
                self.config.min_revenue_growth, self.config.max_revenue_growth,
                self.config.revenue_growth_quarters)

                stock_list, context.previous_date,
                self.config.min_profit_growth, self.config.max_profit_growth,
                self.config.profit_growth_quarters)
        
        if self.config.use_composite_growth:
            stock_list = self.growth_screener.screen_by_composite_growth(
                stock_list, context.previous_date, self.config.growth_weights)
        
        # 5. 质量筛选
        if self.config.use_roe_filter:
            if self.config.roe_above_market:
                stock_list = self.quality_screener.screen_by_roe_above_market(
                    stock_list, context.previous_date, self.config.roe_quarters)
            else:
                stock_list = self.quality_screener.screen_by_roe(
                    stock_list, context.previous_date, self.config.min_roe)
        
        if self.config.use_roa_filter:
            stock_list = self.quality_screener.screen_by_roa(
                stock_list, context.previous_date, self.config.min_roa)
        
        if self.config.use_fcf_filter:
            stock_list = self.quality_screener.screen_by_fcf(
                stock_list, context.previous_date, self.config.fcf_years)
        
        if self.config.use_current_ratio:
            stock_list = self.quality_screener.screen_by_current_ratio(
                stock_list, context.previous_date)
        
        # 6. 排序
        self.buy_list = self._sort_stocks(stock_list, context)
        
        log.info(f'满足条件的股票有{len(self.buy_list)}只')
    
    def market_open(self, context):
        """开盘时运行 - 交易"""
        # 趋势过滤(可选)
        if self.config.use_trend_filter:
            if not self._check_trend(context):
                return
        
        # RSRS择时(可选)
        if self.config.use_rsrs_timing:
            signal = self._get_rsrs_signal(context)
            if signal == 'SELL':
                self._sell_all(context)
                return
            elif signal == 'KEEP' and not self.buy_list:
                return
        
        # 卖出不在买入列表的股票
        self._sell(context, self.buy_list)
        
        # 买入
        self._buy(context, self.buy_list)
    
    def check_limit_up(self, context):
        """盘中检查涨停股 - 所有策略通用模式"""
        if not self.high_limit_list:
            return
        
        current_data = get_current_data()
        for stock in self.high_limit_list:
            if current_data[stock].last_price < current_data[stock].high_limit:
                log.info(f"[{stock}]涨停打开，卖出")
                order_target(stock, 0)
            else:
                log.info(f"[{stock}]涨停，继续持有")
    
    def _get_stock_pool(self, context) -> List[str]:
        """获取初始股票池"""
        if self.config.stock_pool == 'hs300':
            return get_index_stocks('000300.XSHG')
        elif self.config.stock_pool == 'zz500':
            return get_index_stocks('000905.XSHG')
        elif self.config.stock_pool == 'all_a':
            return list(get_all_securities(types=['stock']).index)
        else:
            return get_index_stocks(self.config.stock_pool)
    
    def _filter_basic(self, stock_list: List[str], context) -> List[str]:
        """基础过滤 - ST/停牌/次新/科创板"""
        current_data = get_current_data()
        
        # 过滤次新股
        min_date = context.previous_date - timedelta(days=self.config.min_list_days)
        stock_list = [s for s in stock_list if get_security_info(s).start_date < min_date]
        
        # 过滤ST/停牌/涨跌停
        stock_list = [s for s in stock_list if not (
            current_data[s].paused or
            current_data[s].is_st or
            'ST' in current_data[s].name or
            '*' in current_data[s].name or
            '退' in current_data[s].name or
            (self.config.exclude_kcb and s.startswith('688')) or
            (self.config.exclude_cyb and s.startswith('30')) or
            (self.config.exclude_bse and (s.startswith('4') or s.startswith('8')))
        )]
        
        return stock_list
    
    def _sort_stocks(self, stock_list: List[str], context) -> List[str]:
        """排序选股列表"""
        if not stock_list:
            return []
        
        # 按配置排序
        df = get_fundamentals(query(
            valuation.code,
            valuation.circulating_market_cap,
            valuation.pe_ratio
        ).filter(valuation.code.in_(stock_list)), date=context.previous_date)
        
        # 多因子排序
        for factor, direction in self.config.sort_factors.items():
            if factor in df.columns:
                ascending = direction == 'asc'
                df = df.sort_values(factor, ascending=ascending)
        
        return list(df.code)
    
    def _check_trend(self, context) -> bool:
        """趋势过滤 - 均线判断"""
        h = history(self.config.trend_ma_period + 1, '1d', 'close', 
                   [self.config.trend_index], df=True)
        p = h[self.config.trend_index]
        return p[-1] >= p.mean()
    
    def _get_rsrs_signal(self, context) -> str:
        """RSRS择时信号"""
        # 实现RSRS指标计算
        # 返回: BUY/SELL/KEEP
        pass
    
    def _sell(self, context, buy_list: List[str]):
        """卖出不在买入列表的股票"""
        for stock in list(context.portfolio.positions.keys()):
            if stock not in buy_list and stock not in self.high_limit_list:
                current_data = get_current_data()
                if current_data[stock].last_price < current_data[stock].high_limit:
                    order_target_value(stock, 0)
    
    def _buy(self, context, buy_list: List[str]):
        """买入股票"""
        # 截取前N只
        buy_list = buy_list[:self.config.max_stocks]
        
        # 计算每只股票的目标市值
        target_count = self.config.max_stocks - len(context.portfolio.positions)
        if target_count <= 0:
            return
        
        if self.config.position_mode == 'equal':
            value = context.portfolio.cash / target_count
        elif self.config.position_mode == 'volatility_weight':
            weights = self.volatility_calc.get_volatility_weight(
                buy_list, context.previous_date, self.config.volatility_period)
            for stock in buy_list:
                if stock not in context.portfolio.positions:
                    target_value = context.portfolio.total_value * weights.get(stock, 0)
                    order_target_value(stock, target_value)
            return
        
        for stock in buy_list:
            if stock not in context.portfolio.positions:
                order_target_value(stock, value)
                if len(context.portfolio.positions) >= self.config.max_stocks:
                    break
    
    def _sell_all(self, context):
        """清仓"""
        for stock in list(context.portfolio.positions.keys()):
            order_target_value(stock, 0)
```

```python
def neutralize_by_industry(self, df: pd.DataFrame, factor: str, 
                            date: str) -> pd.DataFrame:
    """行业中性化处理"""
    # 获取行业分类
    industry_info = get_industry(list(df.index), date=date)

    )
```
