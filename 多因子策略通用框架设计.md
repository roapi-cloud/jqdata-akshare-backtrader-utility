# 多因子策略通用框架设计

## 一、策略类型概述

### 核心逻辑
多因子策略通过构建多个选股因子，对股票进行综合评分排序，买入排名靠前的股票，定期调仓。核心流程：**股票池过滤 → 因子计算 → 因子处理 → 因子合成 → 排序选股 → 组合构建 → 执行调仓**。

### 因子体系
覆盖9大类因子：规模、价值、成长、质量、动量、波动、情绪、现金流、杠杆。因子来源包括：
- 财务数据（valuation/balance/income/cash_flow/indicator 表）
- 聚宽因子库（get_factor_values API，数百个预计算因子）
- 自定义因子（继承 Factor 类）
- Alpha191 价量因子
- 技术指标（MACD/RSI/BIAS 等）

### 适用场景
- 宽基指数成分股选股（沪深300/中证500/中证800）
- 全A股选股
- 指数增强
- ETF轮动择时

### 3.2 因子处理流程

```
1. 因子计算
   ├── get_factor_values(securities, factors, end_date, count)  # 聚宽预计算因子
   ├── get_fundamentals(query(...))                              # 财务原始数据计算
   └── calc_factors() with custom Factor class                   # 自定义因子

2. 去极值 (Winsorization)
   ├── winsorize_med(factor, scale=5, axis=0)    # MAD法，聚宽API
   ├── winsorize(factor, qrange=[0.05, 0.95])     # 分位数法
   └── MAD: median ± n * MAD, 超出部分截断

3. 标准化 (Standardization)
   ├── standardlize(factor, axis=0)               # Z-Score, 聚宽API
   └── (x - mean) / std

4. 中性化 (Neutralization)
   ├── neutralize(factor, how=['sw_l1', 'market_cap'], date, axis=0)  # 行业+市值中性
   ├── neutralize(factor, how=['jq_l1', 'market_cap'], fillna='sw_l1')
   └── 线性回归取残差: factor ~ industry_dummies + log(market_cap)

5. 缺失值处理
   ├── fillna(mean)                               # 列均值填充
   ├── 行业均值填充                                # 同行业股票均值
   └── dropna()                                   # 直接删除

6. 因子合成
   ├── 等权合成: score = Σ rank(factor_i) * direction_i
   ├── 加权合成: score = Σ w_i * factor_i
   ├── 回归系数加权: w = OLS(X, Y).params
   ├── IC加权: w = IC_i / Σ|IC_i|
   ├── 机器学习预测: model.predict(X)
   └── 分位叠加: 各因子独立截取Top N%后取交集/并集
```

#### 模块2: 因子处理器 (FactorProcessor)

```python
class FactorProcessor:
    """因子预处理: 去极值 → 缺失值 → 标准化 → 中性化"""
    
    def winsorize_mad(self, factor_series, n=3.0):
        """MAD去极值"""
        median = factor_series.median()
        mad = (factor_series - median).abs().median()
        upper = median + n * mad * 1.4826  # 1.4826使MAD与标准差一致
        lower = median - n * mad * 1.4826
        return factor_series.clip(lower, upper)
    
    def winsorize_quantile(self, factor_series, q_low=0.025, q_high=0.975):
        """分位数去极值"""
        lower = factor_series.quantile(q_low)
        upper = factor_series.quantile(q_high)
        return factor_series.clip(lower, upper)
    
    def standardize_zscore(self, factor_series):
        """Z-Score标准化"""
        mean = factor_series.mean()
        std = factor_series.std()
        if std == 0:
            return factor_series * 0
        return (factor_series - mean) / std
    
    def standardize_rank(self, factor_series, ascending=True):
        """排名标准化"""
        return factor_series.rank(ascending=ascending, pct=True)
    
    def neutralize(self, factor_series, industry_series, market_cap_series, date):
        """行业+市值中性化（回归取残差）"""
        df = pd.DataFrame({
            'factor': factor_series,
            'ln_cap': np.log(market_cap_series),
        })
        # 行业虚拟变量
        industries = industry_series.unique()
        for ind in industries:
            df[f'ind_{ind}'] = (industry_series == ind).astype(float)
        
        df = df.dropna()
        if len(df) < 10:
            return factor_series
        
        y = df['factor']
        X = df.drop('factor', axis=1)
        X = sm.add_constant(X)
        
        try:
            model = sm.OLS(y, X).fit()
            resid = model.resid
            return resid.reindex(factor_series.index).fillna(0)
        except:
            return factor_series
    
    def fill_missing(self, factor_series, method='mean'):
        """缺失值填充"""
        if method == 'mean':
            return factor_series.fillna(factor_series.mean())
        elif method == 'median':
            return factor_series.fillna(factor_series.median())
        elif method == 'industry_mean':
            # 需要外部传入行业信息
            return factor_series.fillna(factor_series.mean())
        return factor_series
    
    def process(self, factor_df, industry_series=None, market_cap_series=None, 
                date=None, steps=None):
        """完整处理流程"""
        if steps is None:
            steps = ['winsorize', 'fillna', 'standardize', 'neutralize']
        
        df = factor_df.copy()
        for col in df.columns:
            series = df[col]
            if 'winsorize' in steps:
                series = self.winsorize_mad(series)
            if 'fillna' in steps:
                series = self.fill_missing(series)
            if 'standardize' in steps:
                series = self.standardize_zscore(series)
            if 'neutralize' in steps and industry_series is not None:
                series = self.neutralize(series, industry_series, market_cap_series, date)
            df[col] = series
        return df
```

#### 模块3: 因子合成器 (FactorCombiner)

```python
class FactorCombiner:
    """因子合成: 等权/加权/回归/ML"""
    
    def equal_weight_score(self, factor_df, directions=None):
        """等权打分合成
        directions: dict, {factor_name: 1 or -1}, 1=越大越好, -1=越小越好
        """
        if directions is None:
            directions = {col: 1 for col in factor_df.columns}
        
        score = pd.Series(0.0, index=factor_df.index)
        for col in factor_df.columns:
            direction = directions.get(col, 1)
            rank = factor_df[col].rank(ascending=(direction < 0))
            score += rank
        return score
    
    def weighted_sum(self, factor_df, weights):
        """加权求和
        weights: dict, {factor_name: weight}
        """
        score = pd.Series(0.0, index=factor_df.index)
        for col, w in weights.items():
            if col in factor_df.columns:
                score += w * factor_df[col]
        return score
    
    def regression_weights(self, factor_df, returns):
        """横截面回归求因子权重"""
        X = sm.add_constant(factor_df.dropna())
        y = returns.reindex(X.index)
        model = sm.OLS(y, X).fit()
        return model.params
    
    def predict_by_regression(self, factor_df, weights):
        """用回归权重预测得分"""
        X = sm.add_constant(factor_df)
        return X.dot(weights)
    
    def ml_predict(self, factor_df, model):
        """机器学习模型预测"""
        features = factor_df.dropna()
        if hasattr(model, 'predict_proba'):
            proba = model.predict_proba(features)
            return pd.Series(proba[:, 1], index=features.index)
        return pd.Series(model.predict(features), index=features.index)
    
    def quantile_filter(self, factor_df, thresholds):
        """分位过滤: 逐层截取
        thresholds: list of (factor_name, sort_ascending, percentile)
        """
        stocks = set(factor_df.index)
        for factor_name, ascending, p in thresholds:
            if factor_name not in factor_df.columns:
                continue
            sub = factor_df.loc[list(stocks), factor_name].dropna()
            n_keep = max(1, int(len(sub) * p))
            selected = sub.nsmallest(n_keep) if ascending else sub.nlargest(n_keep)
            stocks = stocks & set(selected.index)
        return list(stocks)
```

#### 模块4: 组合构建器 (PortfolioConstructor)

```python
class PortfolioConstructor:
    """组合构建: TopN/分位/优化"""
    
    def top_n(self, scores, n=50):
        """Top N选股"""
        return scores.nlargest(n).index.tolist()
    
    def quantile_select(self, scores, quantile=0.1):
        """分位数选股"""
        threshold = scores.quantile(1 - quantile)
        return scores[scores >= threshold].index.tolist()
    
    def equal_weight(self, stock_list, total_value):
        """等权分配"""
        if not stock_list:
            return {}
        weight = 1.0 / len(stock_list)
        return {s: weight * total_value for s in stock_list}
    
    def market_cap_weight(self, stock_list, market_cap_series, total_value):
        """市值加权"""
        mc = market_cap_series.reindex(stock_list).dropna()
        weights = mc / mc.sum()
        return {s: w * total_value for s, w in weights.items()}
    
    def optimize_weights(self, stock_list, date, total_value, 
                         max_single_weight=0.1, min_weight=0.01):
        """组合优化器构建权重（调用聚宽optimizer）"""
        from jqlib.optimizer import portfolio_optimizer, MinVariance, WeightConstraint
        optimized = portfolio_optimizer(
            date=date,
            securities=stock_list,
            target=MinVariance(count=250),
            constraints=[WeightConstraint(low=min_weight, high=max_single_weight)],
            return_none_if_fail=True
        )
        if optimized is None:
            return self.equal_weight(stock_list, total_value)
        return {s: w * total_value for s, w in optimized.items()}
```

```python
        self.processor = FactorProcessor()
        self.combiner = FactorCombiner()
        self.portfolio = PortfolioConstructor()
        self.day_count = 0
        
    # ========== 聚宽策略入口 ==========
    def initialize(self, context):
        """初始化"""
        set_benchmark(self.config.benchmark)
        set_option('use_real_price', True)
        set_option('avoid_future_data', True)
        set_slippage(FixedSlippage(self.config.slippage))
        set_order_cost(OrderCost(
            close_tax=self.config.tax_rate,
            open_commission=self.config.commission_rate,
            close_commission=self.config.commission_rate,
            min_commission=5
        ), type='stock')
        log.set_level('order', 'error')
        
        # 根据调仓频率设置运行函数
        if self.config.rebalance_frequency == 'daily':
            run_daily(self.rebalance, time='open')
        elif self.config.rebalance_frequency == 'weekly':
            run_weekly(self.rebalance, weekday=self.config.rebalance_weekday, time='open')
        elif self.config.rebalance_frequency == 'monthly':
            run_monthly(self.rebalance, monthday=self.config.rebalance_monthday, time='open')
        else:
            run_daily(self._check_rebalance, time='open')
    
    def _check_rebalance(self, context):
        """按天数计数调仓"""
        if self.day_count % self.config.rebalance_days == 0:
            self.rebalance(context)
        self.day_count += 1
    
    # ========== 核心调仓流程 ==========
    def rebalance(self, context):
        """完整调仓流程"""
        date = context.previous_date
        
        # Step 1: 获取股票池
        stock_pool = self._get_stock_pool(context, date)
        if not stock_pool:
            return
        
        # Step 2: 计算因子值
        factor_df = self._calculate_factors(stock_pool, date)
        if factor_df.empty:
            return
        
        # Step 3: 因子处理
        factor_df = self._process_factors(factor_df, stock_pool, date)
        
        # Step 4: 因子合成 → 得分
        scores = self._combine_factors(factor_df, context)
        
        # Step 5: 选股
        target_list = self._select_stocks(scores)
        
        # Step 6: 交易过滤
        target_list = self._apply_trading_filters(context, target_list)
        
        # Step 7: 执行调仓
        self._execute_rebalance(context, target_list)
    
    # ========== Step 1: 股票池 ==========
    def _get_stock_pool(self, context, date):
        """获取并过滤股票池"""
        cfg = self.config
        if cfg.stock_pool == 'all_a':
            stocks = get_all_securities(types=['stock'], date=date).index.tolist()
        elif cfg.stock_pool == 'hs300':
            stocks = get_index_stocks('000300.XSHG', date)
        elif cfg.stock_pool == 'zz500':
            stocks = get_index_stocks('399905.XSHE', date)
        elif cfg.stock_pool == 'zz800':
            stocks = get_index_stocks('399906.XSHE', date)
        elif cfg.stock_pool == 'zz1000':
            stocks = get_index_stocks('399852.XSHG', date)
        else:
            stocks = get_all_securities(types=['stock'], date=date).index.tolist()
        
        # 过滤
        if cfg.filter_st:
            stocks = self._filter_st(stocks)
        if cfg.filter_paused:
            stocks = self._filter_paused(stocks)
        if cfg.filter_new_stock_days > 0:
            stocks = self._filter_new_stocks(context, stocks, cfg.filter_new_stock_days)
        if cfg.filter_kcb:
            stocks = [s for s in stocks if not s.startswith('688')]
        if cfg.filter_bse:
            stocks = [s for s in stocks if not s.startswith(('4', '8'))]
        
        return stocks
    
    # ========== Step 2: 因子计算 ==========
    def _calculate_factors(self, stock_list, date):
        """计算所有配置的因子"""
        all_factors = {}
        
        for fc in self.config.factors:
            if fc.source == 'jqfactor':
                values = get_factor_values(stock_list, [fc.name], end_date=date, count=1)
                all_factors[fc.name] = values[fc.name].iloc[0]
            elif fc.source == 'fundamental':
                # 从财务报表计算
                pass  # 调用 FactorLibrary.get_fundamental_factors
        
        return pd.DataFrame(all_factors)
    
    # ========== Step 3: 因子处理 ==========
    def _process_factors(self, factor_df, stock_list, date):
        """因子预处理"""
        industry = None
        market_cap = None
        
        if self.config.neutralize_industry or self.config.neutralize_market_cap:
            # 获取行业和市值数据
            pass
        
        return self.processor.process(
            factor_df,
            industry_series=industry,
            market_cap_series=market_cap,
            date=date,
            steps=self.config.processing_steps
        )
    
    # ========== Step 4: 因子合成 ==========
    def _combine_factors(self, factor_df, context):
        """因子合成得到综合得分"""
        cfg = self.config
        
        if cfg.combine_method == 'equal_weight':
            directions = {fc.name: fc.direction for fc in cfg.factors}
            return self.combiner.equal_weight_score(factor_df, directions)
        
        elif cfg.combine_method == 'weighted_sum':
            weights = cfg.factor_weights or {fc.name: fc.weight for fc in cfg.factors}
            return self.combiner.weighted_sum(factor_df, weights)
        
        elif cfg.combine_method == 'ml':
            # 加载预训练模型
            import pickle
            with open(cfg.ml_model_path, 'rb') as f:
                model = pickle.load(f)
            return self.combiner.ml_predict(factor_df, model)
        
        return pd.Series(0.0, index=factor_df.index)
    
    # ========== Step 5: 选股 ==========
    def _select_stocks(self, scores):
        """根据得分选股"""
        cfg = self.config
        if cfg.selection_method == 'top_n':
            return self.combiner.top_n(scores, cfg.stock_num)
        elif cfg.selection_method == 'quantile':
            return self.combiner.quantile_select(scores, cfg.quantile_threshold)
        return []
    
    # ========== Step 6: 交易过滤 ==========
    def _apply_trading_filters(self, context, stock_list):
        """应用交易层面过滤"""
        if self.config.filter_limit:
            stock_list = self._filter_limit_up(context, stock_list)
            stock_list = self._filter_limit_down(context, stock_list)
        return stock_list
    
    # ========== Step 7: 执行调仓 ==========
    def _execute_rebalance(self, context, target_list):
        """执行调仓: 先卖后买"""
        cfg = self.config
        current_hold = list(context.portfolio.positions.keys())
        
        # 卖出不在目标列表的股票
        for stock in current_hold:
            if stock not in target_list:
                # 涨停股特殊处理
                if cfg.limit_up_hold and self._is_limit_up(stock):
                    continue
                order_target_value(stock, 0)
        
        # 买入目标股票
        if not target_list:
            return
            
        total_value = context.portfolio.total_value
        
        if cfg.weight_method == 'equal_weight':
            allocations = self.portfolio.equal_weight(target_list, total_value)
        elif cfg.weight_method == 'optimize':
            allocations = self.portfolio.optimize_weights(
                target_list, context.previous_date, total_value,
                max_single_weight=cfg.max_single_weight
            )
        else:
            allocations = self.portfolio.equal_weight(target_list, total_value)
        
        for stock, value in allocations.items():
            if stock not in context.portfolio.positions:
                order_target_value(stock, value)
    
    # ========== 辅助过滤函数 ==========
    def _filter_st(self, stock_list):
        current_data = get_current_data()
        return [s for s in stock_list 
                if not current_data[s].is_st 
                and 'ST' not in current_data[s].name 
                and '*' not in current_data[s].name 
                and '退' not in current_data[s].name]
    
    def _filter_paused(self, stock_list):
        current_data = get_current_data()
        return [s for s in stock_list if not current_data[s].paused]
    
    def _filter_new_stocks(self, context, stock_list, days):
        yesterday = context.previous_date
        return [s for s in stock_list 
                if (yesterday - get_security_info(s).start_date).days > days]
    
    def _filter_limit_up(self, context, stock_list):
        last_prices = history(1, unit='1m', field='close', security_list=stock_list)
        current_data = get_current_data()
        return [s for s in stock_list 
                if s in context.portfolio.positions 
                or last_prices[s][-1] < current_data[s].high_limit]
    
    def _filter_limit_down(self, context, stock_list):
        last_prices = history(1, unit='1m', field='close', security_list=stock_list)
        current_data = get_current_data()
        return [s for s in stock_list 
                if s in context.portfolio.positions 
                or last_prices[s][-1] > current_data[s].low_limit]
    
    def _is_limit_up(self, stock):
        current_data = get_current_data()
        return current_data[stock].last_price >= current_data[stock].high_limit
```

```python
def apply_risk_control(self, context, target_list):
    """风控: 止损/止盈/黑名单"""
    cfg = self.config
    
    # 个股止损
    if cfg.stop_loss > 0:
        for stock in list(context.portfolio.positions.keys()):
            cost = context.portfolio.positions[stock].avg_cost
            price = context.portfolio.positions[stock].price
            if (price - cost) / cost < -cfg.stop_loss:

    if cfg.not_buy_again_days > 0:
```

### 8.1 通用股票过滤模板

```python
def filter_stocks(context, stock_list):
    """通用过滤: ST/停牌/涨跌停/次新/科创/北交"""
    current_data = get_current_data()
    yesterday = context.previous_date
    
    result = []
    for stock in stock_list:
        # ST过滤
        if current_data[stock].is_st:
            continue
        if 'ST' in current_data[stock].name or '*' in current_data[stock].name or '退' in current_data[stock].name:
            continue
        # 停牌过滤
        if current_data[stock].paused:
            continue
        # 科创板过滤
        if stock.startswith('688'):
            continue
        # 北交所过滤
        if stock.startswith(('4', '8')):
            continue
        # 次新股过滤
        if (yesterday - get_security_info(stock).start_date).days < 250:
            continue
        result.append(stock)
    return result
```

```python
def rebalance(context, target_list):
    """先卖后买, 等权分配"""
    # 卖出
    for stock in list(context.portfolio.positions.keys()):
        if stock not in target_list:
            order_target_value(stock, 0)
    
    # 买入
    if target_list:
        cash = context.portfolio.cash / len(target_list)
        for stock in target_list:
            if context.portfolio.positions[stock].total_amount == 0:
                order_target_value(stock, cash)
```

### 8.4 因子获取标准模式

```python
def get_factors(stock_list, factor_names, date):
    """标准因子获取+处理流程"""
    # 1. 获取因子值
    raw = get_factor_values(stock_list, factor_names, end_date=date, count=1)
    df = pd.DataFrame(index=stock_list)
    for name in factor_names:
        df[name] = raw[name].iloc[0]
    
    # 2. 去极值
    df = winsorize_med(df, scale=5, axis=0)
    
    # 3. 缺失值填充
    df = df.fillna(df.mean())
    
    # 4. 中性化
    df = neutralize(df, how=['sw_l1', 'market_cap'], date=date, axis=0)
    
    # 5. 标准化
    df = standardlize(df, axis=0)
    
    return df
```
