# 风控/择时策略通用框架设计

## 一、策略类型概述

### 核心逻辑
风控/择时策略通过量化指标判断市场状态（牛市/熊市/震荡），动态调整仓位或切换资产，以规避系统性风险、捕捉趋势收益。核心思想是"顺势而为、逆势减仓"。

### 适用场景
- **宽基指数择时**：沪深300、上证50、中证500、创业板指
- **ETF轮动配合**：动量ETF轮动 + RSRS择时过滤
- **个股策略风控**：小市值/多因子策略 + 大盘择时止损模块
- **资产配置**：股债平衡、全天候策略中的动态权重调整

### 风险特征
- **优势**：显著降低最大回撤（从30%+降至15%以内），提高夏普比率
- **劣势**：震荡市频繁交易产生磨损，参数敏感度高（N/M/threshold）
- **典型参数**：RSRS(N=18, M=600, threshold=0.7)，均线(MA10/MA20/MA90)

#### 模块1: RSRS择时器 (RSRSTimer)

```python
import numpy as np
import pandas as pd
import statsmodels.api as sm
from scipy import stats
from typing import List, Tuple, Dict, Optional

class RSRSTimer:
    """RSRS(Resistance Support Relative Strength)择时器
    
    核心逻辑: 对最高价vs最低价做OLS/WLS回归, 斜率β反映支撑阻力相对强度,
    标准化后乘以R²(或β)得到择时信号.
    """
    
    # RSRS信号类型
    RSRS_BASIC = "rsrs_basic"              # 原始RSRS = β
    RSRS_ZSCORE = "rsrs_zscore"            # 标准分RSRS = z_score(β)
    RSRS_REVISE = "rsrs_revise"            # 修正标准分 = z_score(β) × R²
    RSRS_RIGHT = "rsrs_right"              # 右偏修正 = z_score(β) × R² × β
    RSRS_PASSIVATION = "rsrs_passivation"  # 钝化RSRS = z_score(β) × R^(4×quantile)
    RSRS_VOL_WEIGHTED = "rsrs_vol_weighted" # 成交量加权钝化
    
    def __init__(
        self, 
        n: int = 18,           # 回归窗口
        m: int = 600,          # 标准化窗口
        threshold: float = 0.7, # 买卖阈值
        method: str = "ols",   # ols / wls
        signal_type: str = "rsrs_right"
    ):
        self.n = n
        self.m = m
        self.threshold = threshold
        self.method = method
        self.signal_type = signal_type
        self.slope_history: List[float] = []
        self.r2_history: List[float] = []
        self.volatility_history: List[float] = []
        
    def calc_slope_ols(self, highs: pd.Series, lows: pd.Series) -> Tuple[float, float]:
        """OLS回归计算斜率和R²"""
        X = sm.add_constant(lows)
        model = sm.OLS(highs, X)
        results = model.fit()
        return results.params[1], results.rsquared
    
    def calc_slope_wls(
        self, 
        highs: pd.Series, 
        lows: pd.Series, 
        volumes: pd.Series
    ) -> Tuple[float, float]:
        """成交量加权最小二乘法(WLS)"""
        X = sm.add_constant(lows)
        weights = volumes / volumes.sum()
        model = sm.WLS(highs, X, weights=weights)
        results = model.fit()
        return results.params[1], results.rsquared
    
    def calc_slope_fast(self, highs: pd.Series, lows: pd.Series) -> Tuple[float, float]:
        """快速OLS(使用协方差, 避免sm库开销)"""
        X = np.array(lows)
        y = np.array(highs)
        n = len(y)
        cov = np.cov(X, y)
        beta = cov[0, 1] / cov[0, 0]
        r2 = beta ** 2 * cov[0, 0] / cov[1, 1]
        return beta, r2
    
    def calc_z_score(self, slopes: List[float]) -> float:
        """计算标准化Z-Score"""
        section = slopes[-self.m:]
        mu = np.mean(section)
        sigma = np.std(section)
        if sigma == 0:
            return 0.0
        return (section[-1] - mu) / sigma
    
    def calc_ret_quantile(self, returns: pd.Series) -> float:
        """计算收益率波动率的分位数(钝化所需)"""
        ret_std = returns.rolling(self.n).std()
        ret_quantile = ret_std.rolling(self.m).apply(
            lambda x: x.rank(pct=True).iloc[-1], raw=False
        )
        return ret_quantile.iloc[-1]
    
    def update(self, highs: pd.Series, lows: pd.Series, 
               volumes: Optional[pd.Series] = None,
               returns: Optional[pd.Series] = None) -> float:
        """更新RSRS序列并返回当前信号值"""
        if self.method == "wls" and volumes is not None:
            beta, r2 = self.calc_slope_wls(highs, lows, volumes)
        else:
            beta, r2 = self.calc_slope_ols(highs, lows)
        
        self.slope_history.append(beta)
        self.r2_history.append(r2)
        
        zscore = self.calc_z_score(self.slope_history)
        
        if self.signal_type == self.RSRS_BASIC:
            return beta
        elif self.signal_type == self.RSRS_ZSCORE:
            return zscore
        elif self.signal_type == self.RSRS_REVISE:
            return zscore * r2
        elif self.signal_type == self.RSRS_RIGHT:
            return zscore * beta * r2
        elif self.signal_type == self.RSRS_PASSIVATION:
            if returns is not None:
                quantile = self.calc_ret_quantile(returns)
                return zscore * (r2 ** (4 * quantile))
            return zscore * r2
        elif self.signal_type == self.RSRS_VOL_WEIGHTED:
            if returns is not None:
                quantile = self.calc_ret_quantile(returns)
                return zscore * (r2 ** (2 * quantile))
            return zscore * r2
        return zscore * r2
    
    def get_signal(self, score: float) -> int:
        """获取择时信号: 1=买入, 0=持有, -1=卖出"""
        if score > self.threshold:
            return 1   # BUY
        elif score < -self.threshold:
            return -1  # SELL
        else:
            return 0   # KEEP
```

#### 模块2: 均线择时器 (MATimer)

```python
class MATimer:
    """均线/趋势择时器
    
    支持: 单均线突破、双均线交叉、乖离率、布林带
    """
    
    def __init__(
        self,
        method: str = "single_ma",  # single_ma / dual_ma / bias / boll
        period: int = 10,
        fast_period: int = 5,
        slow_period: int = 20,
        threshold: float = 0.003,
        boll_std: float = 2.0
    ):
        self.method = method
        self.period = period
        self.fast_period = fast_period
        self.slow_period = slow_period
        self.threshold = threshold
        self.boll_std = boll_std
    
    def get_signal_single_ma(self, close: pd.Series) -> int:
        """单均线突破: 价格 > MA(period) 买入"""
        ma = close.rolling(self.period).mean()
        if close.iloc[-1] > ma.iloc[-1] * (1 + self.threshold):
            return 1
        elif close.iloc[-1] < ma.iloc[-1] * (1 - self.threshold):
            return -1
        return 0
    
    def get_signal_dual_ma(self, close: pd.Series) -> int:
        """双均线交叉: 快线上穿慢线买入"""
        fast_ma = close.rolling(self.fast_period).mean()
        slow_ma = close.rolling(self.slow_period).mean()
        if fast_ma.iloc[-1] > slow_ma.iloc[-1] and fast_ma.iloc[-2] <= slow_ma.iloc[-2]:
            return 1
        elif fast_ma.iloc[-1] < slow_ma.iloc[-1] and fast_ma.iloc[-2] >= slow_ma.iloc[-2]:
            return -1
        return 0
    
    def get_signal_bias(self, close: pd.Series) -> float:
        """乖离率: (close - MA) / MA"""
        ma = close.rolling(self.period).mean()
        return (close.iloc[-1] / ma.iloc[-1]) - 1
    
    def get_signal_boll(self, close: pd.Series) -> Tuple[float, float, float]:
        """布林带: 返回(mid, upper, lower)"""
        mid = close.rolling(self.period).mean()
        std = close.rolling(self.period).std()
        upper = mid + self.boll_std * std
        lower = mid - self.boll_std * std
        return mid.iloc[-1], upper.iloc[-1], lower.iloc[-1]
    
    def get_signal(self, close: pd.Series) -> int:
        """统一信号接口"""
        if self.method == "single_ma":
            return self.get_signal_single_ma(close)
        elif self.method == "dual_ma":
            return self.get_signal_dual_ma(close)
        elif self.method == "bias":
            bias = self.get_signal_bias(close)
            return 1 if bias > self.threshold else (-1 if bias < -self.threshold else 0)
        elif self.method == "boll":
            mid, upper, lower = self.get_signal_boll(close)
            if close.iloc[-1] > upper:
                return 1
            elif close.iloc[-1] < lower:
                return -1
            return 0
        return 0
```

```python
class MarketBreadthCalculator:
    """市场宽度指标计算器
    
    衡量市场参与度, 宽度收缩预示趋势衰竭
    """
    
    def __init__(self, universe: List[str] = None):
        self.universe = universe  # 股票池, 默认全A
    
    def get_new_high_ratio(
        self, 
        stocks: List[str], 
        date: str, 
        lookback: int = 252,
        get_price_func=None
    ) -> float:
        """创新高比例 = N日内创新高股票数 / 总股票数"""
        count = 0
        for stock in stocks:
            highs = get_price_func(stock, end_date=date, count=lookback, fields='high')
            if highs['high'].iloc[-1] >= highs['high'].max():
                count += 1
        return count / len(stocks) if stocks else 0.0
    
    def get_above_ma_ratio(
        self, 
        stocks: List[str], 
        date: str, 
        ma_period: int = 20,
        get_price_func=None
    ) -> float:
        """站上均线比例"""
        count = 0
        for stock in stocks:
            close = get_price_func(stock, end_date=date, count=ma_period+1, fields='close')
            ma = close['close'].rolling(ma_period).mean().iloc[-1]
            if close['close'].iloc[-1] > ma:
                count += 1
        return count / len(stocks) if stocks else 0.0
    
    def get_bias_ratio(
        self, 
        stocks: List[str], 
        date: str, 
        bias_period: int = 20,
        get_price_func=None
    ) -> float:
        """BIAS>0比例(20日乖离率大于0的股票占比)"""
        count = 0
        for stock in stocks:
            close = get_price_func(stock, end_date=date, count=bias_period+1, fields='close')
            ma = close['close'].rolling(bias_period).mean().iloc[-1]
            bias = close['close'].iloc[-1] / ma - 1
            if bias > 0:
                count += 1
        return count / len(stocks) if stocks else 0.0
    
    def get_diffusion_index(
        self,
        stocks: List[str],
        date: str,
        roc_period: int = 100,
        weight_method: str = "avg",  # avg / mktcap
        get_price_func=None,
        get_valuation_func=None
    ) -> float:
        """扩散指标: ROC>0的成分股占比(可市值加权)"""
        roc_positive = []
        for stock in stocks:
            now = get_price_func(stock, end_date=date, count=1, fields='close')
            last = get_price_func(stock, end_date=date, count=roc_period, fields='close')
            if len(last) >= roc_period and last['close'].iloc[0] > 0:
                roc = now['close'].iloc[-1] / last['close'].iloc[0] - 1
                roc_positive.append(roc > 0)
        
        if weight_method == "mktcap":
            # 市值加权版本
            caps = get_valuation_func(stocks, end_date=date, fields='circulating_market_cap')
            weights = caps['circulating_market_cap'] / caps['circulating_market_cap'].sum()
            return np.sum(np.array(roc_positive) * weights)
        return np.mean(roc_positive)
    
    def get_breadth_signal(
        self, 
        ratio: float, 
        upper_threshold: float = 0.8, 
        lower_threshold: float = 0.2
    ) -> int:
        """市场宽度信号"""
        if ratio > upper_threshold:
            return 1   # 过热, 谨慎
        elif ratio < lower_threshold:
            return -1  # 过冷, 可能见底
        return 0
```

```python
class CrowdingCalculator:
    """拥挤率指标: 衡量资金集中度, 高拥挤率预示顶部风险
    
    拥挤率 = 前5%成交额股票的成交额 / 全市场总成交额
    """
    
    def __init__(self, top_pct: float = 0.05):
        self.top_pct = top_pct  # 前5%
    
    def get_crowding_ratio(
        self, 
        stocks: List[str], 
        date: str,
        get_price_func=None
    ) -> float:
        """计算当日拥挤率"""
        money_data = get_price_func(stocks, end_date=date, count=1, fields='money')
        money_data = money_data.sort_values('money', ascending=False)
        
        n_top = max(1, int(len(money_data) * self.top_pct))
        top_money = money_data['money'].iloc[:n_top].sum()
        total_money = money_data['money'].sum()
        
        return top_money / total_money if total_money > 0 else 0.0
    
    def get_crowding_series(
        self,
        stocks: List[str],
        start_date: str,
        end_date: str,
        get_price_func=None,
        get_trade_days_func=None
    ) -> pd.Series:
        """计算历史拥挤率序列"""
        trade_days = get_trade_days_func(start_date=start_date, end_date=end_date)
        crowd_dict = {}
        
        for date in trade_days:
            crowd_dict[date] = self.get_crowding_ratio(stocks, date, get_price_func)
        
        return pd.Series(crowd_dict, name='crowding_ratio')
    
    def get_top_bottom_signal(
        self, 
        crowd_ratio: float,
        history: pd.Series,
        upper_quantile: float = 0.9,
        lower_quantile: float = 0.1
    ) -> int:
        """顶底信号: 拥挤率处于历史高位=顶部风险, 低位=底部机会"""
        upper = history.quantile(upper_quantile)
        lower = history.quantile(lower_quantile)
        
        if crowd_ratio > upper:
            return -1  # 顶部信号(减仓)
        elif crowd_ratio < lower:
            return 1   # 底部信号(加仓)
        return 0
```

#### 模块5: 波动率平衡器 (VolatilityBalancer)

```python
class VolatilityBalancer:
    """波动率平衡: 基于波动率倒数分配权重(风险平价思想)
    
    weight_i = (1/σ²_i) / Σ(1/σ²_j)
    """
    
    def __init__(self, lookback: int = 40, annualize_factor: float = 250.0):
        self.lookback = lookback
        self.annualize_factor = annualize_factor
    
    def get_volatility(
        self, 
        close: pd.Series, 
        down_only: bool = False,
        clip_3sigma: bool = True
    ) -> float:
        """计算年化波动率"""
        returns = np.log(close / close.shift(1)).dropna()
        
        if down_only:
            returns = returns.clip(upper=0)  # 只考虑下行波动
        
        if clip_3sigma:
            mean = returns.mean()
            std = returns.std()
            returns = returns.clip(mean - 3*std, mean + 3*std)
        
        vol = returns.std() * np.sqrt(self.annualize_factor) * 100
        return vol
    
    def calc_risk_parity_weights(
        self, 
        close_df: pd.DataFrame
    ) -> pd.Series:
        """风险平价权重(波动率平方反比)"""
        volatilities = {}
        for col in close_df.columns:
            volatilities[col] = self.get_volatility(close_df[col])
        
        vol_series = pd.Series(volatilities)
        inv_var = 1.0 / (vol_series ** 2)
        weights = inv_var / inv_var.sum()
        return weights
    
    def get_target_allocation(
        self, 
        close_df: pd.DataFrame,
        target_vol: float = 0.10
    ) -> Dict[str, float]:
        """目标波动率配置"""
        weights = self.calc_risk_parity_weights(close_df)
        portfolio_vol = np.sqrt(np.sum((weights ** 2) * (close_df.pct_change().std() ** 2)))
        
        # 缩放至目标波动率
        scale = target_vol / portfolio_vol if portfolio_vol > 0 else 1.0
        return {k: v * scale for k, v in weights.items()}
```

```python
class BullBearIndicator:
    """波动率与换手率构建的牛熊指标
    
    牛熊指标 = σ(250d) / Turnover(250d)
    与指数走势负相关: 指标上升=熊市, 指标下降=牛市
    """
    
    def __init__(self, periods: int = 250):
        self.periods = periods
    
    def calc_volatility(self, close: pd.Series) -> pd.Series:
        """N日滚动波动率"""
        returns = close.pct_change()
        return returns.rolling(self.periods).std()
    
    def calc_turnover_ma(self, turnover: pd.Series) -> pd.Series:
        """N日滚动平均换手率"""
        return turnover.rolling(self.periods).mean()
    
    def calc_bull_bear_index(
        self, 
        close: pd.Series, 
        turnover: pd.Series
    ) -> pd.Series:
        """计算牛熊指标 = 波动率 / 换手率"""
        vol = self.calc_volatility(close)
        turnover_ma = self.calc_turnover_ma(turnover)
        return vol / turnover_ma
    
    def get_signal(
        self, 
        bb_index: pd.Series,
        lookback: int = 60
    ) -> int:
        """牛熊信号: 指标趋势上升=熊, 下降=牛"""
        recent = bb_index.tail(lookback)
        slope = np.polyfit(np.arange(len(recent)), recent.values, 1)[0]
        
        if slope > 0.001:
            return -1  # 熊市信号
        elif slope < -0.001:
            return 1   # 牛市信号
        return 0
```

```python
class NorthboundTimer:
    """北向资金布林带择时
    
    利用沪深港通净流入的布林带突破判断市场方向
    """
    
    def __init__(
        self,
        window: int = 252,
        stdev_n: float = 1.5
    ):
        self.window = window
        self.stdev_n = stdev_n
    
    def get_boll_bands(
        self, 
        net_amounts: pd.Series
    ) -> Tuple[float, float, float]:
        """计算北向资金布林带"""
        recent = net_amounts.tail(self.window)
        mid = recent.mean()
        std = recent.std()
        upper = mid + self.stdev_n * std
        lower = mid - self.stdev_n * std
        return mid, upper, lower
    
    def get_signal(
        self, 
        net_amounts: pd.Series
    ) -> int:
        """北向资金信号: 突破上轨=买, 跌破下轨=卖"""
        mid, upper, lower = self.get_boll_bands(net_amounts)
        latest = net_amounts.iloc[-1]
        
        if latest > upper:
            return 1   # 大幅净流入, 买入
        elif latest < lower:
            return -1  # 大幅净流出, 卖出
        return 0
```

```python
class StockFilter:
    """综合股票过滤器
    
    过滤ST/停牌/新股/科创板/北交所/涨跌停/退市
    优化: 层层过滤缩小股票池, 减少API调用
    """
    
    def __init__(self):
        pass
    
    def filter_st(self, stock_list: List[str], current_data: Dict) -> List[str]:
        """过滤ST股票(覆盖ST/*ST/退)"""
        return [s for s in stock_list if not current_data[s].is_st]
    
    def filter_paused(self, stock_list: List[str], current_data: Dict) -> List[str]:
        """过滤停牌股票"""
        return [s for s in stock_list if not current_data[s].paused]
    
    def filter_delisting(self, stock_list: List[str], current_data: Dict) -> List[str]:
        """过滤退市股票"""
        return [s for s in stock_list if '退' not in current_data[s].name]
    
    def filter_new_stock(
        self, 
        stock_list: List[str], 
        date: str, 
        days: int = 375,
        get_security_info_func=None
    ) -> List[str]:
        """过滤次新股"""
        start_date = date - timedelta(days=days)
        return [
            s for s in stock_list 
            if get_security_info_func(s).start_date <= start_date
        ]
    
    def filter_kcb(self, stock_list: List[str]) -> List[str]:
        """过滤科创板(688开头)"""
        return [s for s in stock_list if not s.startswith('688')]
    
    def filter_bse(self, stock_list: List[str]) -> List[str]:
        """过滤北交所(4/8开头)"""
        return [s for s in stock_list if not (s.startswith('4') or s.startswith('8'))]
    
    def filter_limit_up(
        self, 
        stock_list: List[str], 
        current_data: Dict,
        last_prices: Dict
    ) -> List[str]:
        """过滤涨停股票(已持有的除外)"""
        return [
            s for s in stock_list 
            if last_prices[s] < current_data[s].high_limit
        ]
    
    def filter_limit_down(
        self, 
        stock_list: List[str], 
        current_data: Dict,
        last_prices: Dict
    ) -> List[str]:
        """过滤跌停股票"""
        return [
            s for s in stock_list 
            if last_prices[s] > current_data[s].low_limit
        ]
    
    def filter_all(
        self, 
        stock_list: List[str], 
        date: str,
        current_data: Dict,
        config: Dict = None,
        get_security_info_func=None
    ) -> List[str]:
        """综合过滤(按顺序层层过滤, 提高效率)"""
        if config is None:
            config = {
                'filter_st': True,
                'filter_paused': True,
                'filter_delisting': True,
                'filter_new_stock': True,
                'new_stock_days': 375,
                'filter_kcb': True,
                'filter_bse': True,
            }
        
        # 1. 先过滤代码前缀(无需API)
        if config.get('filter_kcb', True):
            stock_list = self.filter_kcb(stock_list)
        if config.get('filter_bse', True):
            stock_list = self.filter_bse(stock_list)
        
        # 2. 过滤新股(需要info API)
        if config.get('filter_new_stock', True):
            stock_list = self.filter_new_stock(
                stock_list, date, config.get('new_stock_days', 375),
                get_security_info_func
            )
        
        # 3. 过滤ST/停牌/退市(需要current_data API)
        if config.get('filter_st', True):
            stock_list = self.filter_st(stock_list, current_data)
        if config.get('filter_paused', True):
            stock_list = self.filter_paused(stock_list, current_data)
        if config.get('filter_delisting', True):
            stock_list = self.filter_delisting(stock_list, current_data)
        
        return stock_list
```

```python
class RiskTimingModule:
    """风控/择时模块主类
    
    整合所有择时器和风控器, 提供统一接口
    """
    
    def __init__(self, config: RiskTimingConfig):
        self.config = config
        
        # 初始化择时器
        self.rsrs_timer = RSRSTimer(
            n=config.rsrs_config.n,
            m=config.rsrs_config.m,
            threshold=config.rsrs_config.threshold,
            method=config.rsrs_config.method,
            signal_type=config.rsrs_config.signal_type
        )
        self.ma_timer = MATimer(
            method=config.ma_config.method,
            period=config.ma_config.period,
            fast_period=config.ma_config.fast_period,
            slow_period=config.ma_config.slow_period,
            threshold=config.ma_config.threshold
        )
        self.breadth_calc = MarketBreadthCalculator()
        self.crowding_calc = CrowdingCalculator()
        self.vol_balancer = VolatilityBalancer()
        self.bull_bear = BullBearIndicator()
        self.northbound_timer = NorthboundTimer()
        self.stock_filter = StockFilter()
        
        # 状态
        self.current_position_ratio = 1.0
        self.is_holding = False
    
    def get_timing_signal(
        self, 
        date: str,
        index_data: pd.DataFrame,
        northbound_data: Optional[pd.Series] = None,
        market_data: Optional[Dict] = None
    ) -> int:
        """获取综合择时信号
        
        Returns:
            1 = 买入/加仓, 0 = 持有, -1 = 卖出/减仓
        """
        signals = []
        
        # 1. RSRS信号
        if self.config.timing_method in (TimingMethod.RSRS, TimingMethod.COMBINED):
            highs = index_data['high'].tail(self.rsrs_timer.n)
            lows = index_data['low'].tail(self.rsrs_timer.n)
            volumes = index_data.get('volume', None)
            if volumes is not None:
                volumes = volumes.tail(self.rsrs_timer.n)
            returns = index_data['close'].pct_change().tail(self.rsrs_timer.m + self.rsrs_timer.n)
            
            score = self.rsrs_timer.update(highs, lows, volumes, returns)
            signals.append(self.rsrs_timer.get_signal(score))
        
        # 2. 均线信号
        if self.config.timing_method in (TimingMethod.MA, TimingMethod.COMBINED):
            signals.append(self.ma_timer.get_signal(index_data['close']))
        
        # 3. 北向资金过滤
        if self.config.use_northbound_filter and northbound_data is not None:
            nb_signal = self.northbound_timer.get_signal(northbound_data)
            if nb_signal == -1:
                return -1  # 北向大幅流出, 强制卖出
        
        # 4. 市场宽度过滤
        if self.config.use_breadth_filter and market_data:
            breadth = market_data.get('breadth_ratio', 0.5)
            if breadth < 0.1:
                signals.append(1)   # 极度悲观, 可能见底
            elif breadth > 0.9:
                signals.append(-1)  # 极度乐观, 可能见顶
        
        # 综合信号(多数决)
        if not signals:
            return 0
        return 1 if sum(signals) > 0 else (-1 if sum(signals) < 0 else 0)
    
    def get_position_ratio(
        self, 
        date: str,
        timing_signal: int,
        assets_close: Optional[pd.DataFrame] = None
    ) -> float:
        """获取目标仓位比例"""
        pc = self.config.position_control
        
        if timing_signal == 1:
            target = pc.bull_position_ratio
        elif timing_signal == -1:
            target = pc.bear_position_ratio
        else:
            target = (pc.bull_position_ratio + pc.bear_position_ratio) / 2
        
        # 波动率平衡调整
        if pc.use_volatility_balance and assets_close is not None:
            weights = self.vol_balancer.calc_risk_parity_weights(assets_close)
            # 根据择时信号调整总仓位
            return {k: v * target for k, v in weights.items()}
        
        return target
    
    def filter_stocks(
        self, 
        stock_list: List[str], 
        date: str,
        current_data: Dict,
        get_security_info_func=None
    ) -> List[str]:
        """过滤股票池"""
        return self.stock_filter.filter_all(
            stock_list, date, current_data,
            config={
                'filter_st': self.config.risk_control.filter_st,
                'filter_paused': self.config.risk_control.filter_paused,
                'filter_new_stock': self.config.risk_control.filter_new_stock,
                'new_stock_days': self.config.risk_control.new_stock_days,
                'filter_kcb': self.config.risk_control.filter_kcb,
                'filter_bse': self.config.risk_control.filter_bse,
                'filter_delisting': self.config.risk_control.filter_delisting,
            },
            get_security_info_func=get_security_info_func
        )
    
    def check_intraday_stop_loss(
        self,
        position: Dict,
        get_price_func=None
    ) -> List[str]:
        """盘中止损检查: 60分钟跌破MA20则卖出"""
        rc = self.config.risk_control
        if not rc.use_intraday_stop_loss:
            return []
        
        to_sell = []
        for stock in position:
            data = get_price_func(stock, count=rc.intraday_ma_period+2, freq=rc.intraday_freq, fields='close')
            ma = data['close'].rolling(rc.intraday_ma_period).mean()
            if data['close'].iloc[-1] < ma.iloc[-1]:
                to_sell.append(stock)
        return to_sell
```

```python
def lowpass_filter(signal: pd.Series, cutoff: float = 0.1) -> pd.Series:
    """简单低通滤波, 去除高频噪声"""
    alpha = cutoff
    filtered = pd.Series(index=signal.index)
    filtered.iloc[0] = signal.iloc[0]
    for i in range(1, len(signal)):
        filtered.iloc[i] = alpha * signal.iloc[i] + (1 - alpha) * filtered.iloc[i-1]
    return filtered
```

### 5.5 回测性能优化建议

1. **前序数据预加载**: RSRS的M日历史斜率在initialize中一次性计算, 避免每日重复
2. **向量化计算**: 使用`np.polyfit`替代`sm.OLS`, 速度提升10倍+
3. **缓存机制**: 缓存`get_current_data()`结果, 同一天内多次调用只查询一次
4. **分批查询**: 拥挤率/市场宽度等全市场计算, 按100天分批避免内存溢出
5. **并行计算**: 扩散指标中各股票ROC计算可并行化
